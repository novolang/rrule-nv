
// A field of `YYYYMMDDTHHMMSS` as a number.
fn field(text: Str, from: Int, to: Int) -> Int
    str.to_int(str.slice(text, from, to)) ?? 0

// A civil date-time from `YYYYMMDDTHHMMSS`.
fn at(text: Str) -> civil.CivilDateTime
    let d = CivilDate { year: field(text, 0, 4), month: field(text, 4, 6), day: field(text, 6, 8) }
    let t = CivilTime { hour: field(text, 9, 11),
        minute: field(text, 11, 13),
        second: field(text, 13, 15),
        nano: 0 }
    civil.datetime(d, t)

// `n` in at least `width` digits.
fn pad(n: Int, width: Int) -> Str
    str.pad_left("${n}", width, "0")

// The occurrences as the table writes them, comma-separated.
fn stamps(xs: [civil.CivilDateTime]) -> Str
    var out: [Str] = []
    for x in xs
        let d = x.date
        let t = x.time
        list.push(out,
            "${pad(d.year, 4)}${pad(d.month, 2)}${pad(d.day, 2)}T${pad(t.hour, 2)}${pad(t.minute, 2)}${pad(t.second, 2)}")
    str.join(out, ",")

@test
fn test_the_first_occurrences_agree_with_dateutil()
    for line in str.split(vectors(), "\n")
        let f = str.split(line, "|")
        test.case("${f[0]} ${f[1]}")
        match rruleparse.parse(f[1])
            Err(e) => test.fail("${f[1]}: ${rruleerror.code(e)}")
            Ok(r)  =>
                let s = rruleset.with_rule(rruleset.set(at(f[0]), false), r)
                let want = str.split(f[2], ",")
                let got = rruleiter.between(s, at(f[0]), at(want[-1]))
                test.assert_eq(stamps(got), f[2])
